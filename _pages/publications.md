<script>
  (function () {
    // Insert a year heading wherever the year changes in the by-year list
    var lastYear = null;
    document.querySelectorAll("#by-year ol.bibliography > li").forEach(function (li) {
      var row = li.querySelector("[data-year]");
      if (!row) return;
      var year = row.getAttribute("data-year");
      if (year !== lastYear) {
        var heading = document.createElement("li");
        heading.className = "year-heading";
        heading.innerHTML = '<h2 class="bibliography">' + year + "</h2>";
        li.parentNode.insertBefore(heading, li);
        lastYear = year;
      }
    });

    // Toggle between views
    var buttons = document.querySelectorAll(".view-toggle .btn");
    function show(view) {
      document.querySelectorAll(".pub-view").forEach(function (el) {
        el.classList.toggle("active", el.id === view);
      });
      buttons.forEach(function (b) {
        b.classList.toggle("active", b.dataset.view === view);
      });
      try { localStorage.setItem("pubView", view); } catch (e) {}
    }
    buttons.forEach(function (b) {
      b.addEventListener("click", function () { show(b.dataset.view); });
    });
    var saved = null;
    try { saved = localStorage.getItem("pubView"); } catch (e) {}
    if (saved === "by-year" || saved === "by-topic") show(saved);
  })();
</script>
