<script>
  (function () {
    // Topics, in the order the buttons appear. Names must match the category values in papers.bib.
    var TOPICS = [
      ["Social and Environmental Impact", "Research on humanitarian operations, disaster relief, nonprofit management, sustainability, and public safety."],
      ["Revenue Management", "Research on pricing, assortment optimization, and choice modeling."],
      ["Digital Commerce", "Research on e-commerce, online marketplaces, and omnichannel retail."],
      ["New Products and Innovation", "Research on new product introduction, product development, and customer feedback."],
      ["Optimization under Uncertainty", "Research on data-driven and distributionally robust decision-making when distributions or parameters are unknown."]
    ];

    var papers = Array.prototype.slice.call(document.querySelectorAll("#pub-list ol.bibliography > li"));
    function topicsOf(li) {
      var row = li.querySelector("[data-topics]");
      if (!row) return [];
      return row.getAttribute("data-topics").split(";").map(function (t) { return t.trim(); });
    }

    // Insert a year heading wherever the year changes
    var lastYear = null;
    papers.forEach(function (li) {
      var row = li.querySelector("[data-year]");
      if (!row) return;
      var year = row.getAttribute("data-year");
      if (year !== lastYear) {
        var heading = document.createElement("li");
        heading.className = "year-heading";
        heading.innerHTML = '<h2 class="bibliography">' + year + "</h2>";
        li.parentNode.insertBefore(heading, li);
        lastYear = year;
      }
    });

    // Filter buttons with paper counts
    var pillBar = document.getElementById("topic-pills");
    var desc = document.getElementById("topic-desc");
    function addPill(label, key, count) {
      var b = document.createElement("button");
      b.type = "button";
      b.className = "topic-pill" + (key === "all" ? " active" : "");
      b.dataset.key = key;
      b.innerHTML = label + '<span class="count">' + count + "</span>";
      b.addEventListener("click", function () { filterTopic(key); });
      pillBar.appendChild(b);
    }
    function filterTopic(key) {
      // Show or hide papers
      papers.forEach(function (li) {
        li.style.display = (key === "all" || topicsOf(li).indexOf(key) !== -1) ? "" : "none";
      });
      // Hide year headings that have no visible papers under them
      document.querySelectorAll("#pub-list li.year-heading").forEach(function (h) {
        var any = false, el = h.nextElementSibling;
        while (el && !el.classList.contains("year-heading")) {
          if (el.style.display !== "none") { any = true; break; }
          el = el.nextElementSibling;
        }
        h.style.display = any ? "" : "none";
      });
      // Update buttons and description
      pillBar.querySelectorAll(".topic-pill").forEach(function (p) {
        p.classList.toggle("active", p.dataset.key === key);
      });
      var match = TOPICS.filter(function (t) { return t[0] === key; })[0];
      desc.textContent = match ? match[1] : "";
    }

    addPill("All", "all", papers.length);
    TOPICS.forEach(function (t) {
      var n = papers.filter(function (li) { return topicsOf(li).indexOf(t[0]) !== -1; }).length;
      addPill(t[0], t[0], n);
    });
  })();
</script>
